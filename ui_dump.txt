const CLERK_PUBLISHABLE_KEY = typeof window !== 'undefined' 
  ? (process.env.NEXT_PUBLIC_CLERK_PUBLISHABLE_KEY || '') 
  : '';

function MyApp({ Component, pageProps }: any) {
  if (!CLERK_PUBLISHABLE_KEY) {
    return <Component {...pageProps} />;
  }

  return (
    <ClerkProvider publishableKey={CLERK_PUBLISHABLE_KEY}>
      <Component {...pageProps} />
    </ClerkProvider>
  );
}

  const [logFilter, setLogFilter] = useState<'all' | 'alert' | 'telemetry' | 'scada'>('all');

    const time = new Date().toLocaleTimeString();

    try {
      if (!audioCtxRef.current) {
        audioCtxRef.current = new (window.AudioContext || (window as any).webkitAudioContext)();
      }
      const ctx = audioCtxRef.current;
      if (ctx.state === 'suspended') {
        ctx.resume();
      }
      setTimeout(() => {
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(freq, ctx.currentTime);
        gain.gain.setValueAtTime(0.06, ctx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + duration);
        osc.connect(gain);
        gain.connect(ctx.destination);
        osc.start();
        osc.stop(ctx.currentTime + duration);
      }, delay);
    } catch (e) {
      console.log('Audio feedback uninitialized:', e);
    }
  };

  useEffect(() => {
    async function initApp() {
      try {
        setLoading(true);
        await fetch(`${BACKEND_URL}/seed`, { method: 'POST' });
        addLog('Database initialized and PostGIS geometries seeded.', 'scada');

        const res = await fetch(`${BACKEND_URL}/lines`);
        const data = await res.json();
        setLines(data);
        addLog(`Synced ${data.length} transmission line loops from PostGIS.`, 'scada');
      } catch (err) {
        console.error(err);
        addLog('API network failure. Please verify backend service on port 8000.', 'alert');
      } finally {
        setLoading(false);
      }
